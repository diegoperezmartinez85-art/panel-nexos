/** * ============================================================ * NEXOS - RECEPTOR WEB APP * Versión con sistema de BANEO DE VECINOS * ============================================================ */ // ── CONFIGURACIÓN ────────────────────────────────────────── var NOMBRE_HOJA = "Hoja 1"; var NOMBRE_PANICO = "PANICO"; var NOMBRE_BANEADOS = "BANEADOS"; var FILA_INICIO = 12; var COL_FECHA = 4; // D var COL_TIPO = 5; // E var COL_MZA = 7; // G var COL_CASA = 8; // H var COL_DNI = 11; // K var COL_PATENTE = 13; // M var COL_ENTRADA = 14; // N var COL_SALIDA = 15; // O var COL_ID = 16; // P // ─────────────────────────────────────────────────────────── // Columnas hoja PANICO: // A(1)=ID B(2)=Fecha C(3)=Mza D(4)=Casa E(5)=Estado F(6)=MapsURL G(7)=Lat H(8)=Lng // Columnas hoja BANEADOS: // A(1)=MZA B(2)=CASA C(3)=ID_RESIDENTE D(4)=MOTIVO E(5)=FECHA_BAN F(6)=TIMESTAMP /** * GET - Devuelve activos + alertas de pánico pendientes + lista de baneados. */ function doGet(e) { try { var params = e ? (e.parameter || {}) : {}; // ── BÚSQUEDA HISTÓRICA ──────────────────────────────── if (params.accion === 'buscar') { var resultados = buscarHistorico(params.fecha || '', params.q || ''); return jsonOutput({ resultados: resultados }); } // ── LISTA DE BANEADOS (para la app de vecinos) ──────── if (params.accion === 'get_baneados') { var baneados = obtenerBaneados(); return jsonOutput({ baneados: baneados }); } // ── DEFAULT: activos + pánico + baneados (sync normal) ─ var activos = obtenerActivos(); var panico = obtenerPanicoPendiente(); var baneados = obtenerBaneados(); return jsonOutput({ activos: activos, panico: panico, baneados: baneados }); } catch (err) { return jsonOutput({ ok: false, error: err.message }); } } /** * POST - Distingue entre: registro normal, pánico, baneo, desbaneo. */ function doPost(e) { try { var data = JSON.parse(e.postData.contents); // ── PÁNICO ───────────────────────────────────────────── if (data.accion === "panico") { // Verificar si el vecino está baneado antes de registrar if (esBaneado(data.mza, data.casa)) { return jsonOutput({ ok: false, bloqueado: true, msg: "Bloqueado por administración" }); } grabarPanico(data); return jsonOutput({ ok: true, msg: "Alerta enviada" }); } if (data.accion === "ack_panico") { confirmarPanico(data.panicoId); return jsonOutput({ ok: true, msg: "Pánico confirmado" }); } // ── BANEO ────────────────────────────────────────────── if (data.accion === "banear") { grabarBaneo(data); return jsonOutput({ ok: true, msg: "Vecino baneado" }); } if (data.accion === "desbanear") { eliminarBaneo(data.mza, data.casa); return jsonOutput({ ok: true, msg: "Vecino desbaneado" }); } // ── REGISTRO NORMAL ──────────────────────────────────── var patente = (data.pat || "").toString().trim().toUpperCase(); if (!patente) return jsonOutput({ ok: false, error: "Patente vacía" }); if (data.hS) { marcarSalida(data); return jsonOutput({ ok: true, msg: "Salida registrada" }); } else { escribirEntrada(data); return jsonOutput({ ok: true, msg: "Entrada registrada" }); } } catch (err) { return jsonOutput({ ok: false, error: err.message }); } } // ══════════════════════════════════════════════════════════ // ── SISTEMA DE BANEO ────────────────────────────────────── // ══════════════════════════════════════════════════════════ /** * Verifica si una manzana+casa está en la lista negra de baneados. * @returns {boolean} */ function esBaneado(mza, casa) { var baneados = obtenerBaneados(); var mzaN = String(mza || '').trim(); var casaN = String(casa || '').trim(); if (!mzaN || !casaN) return false; for (var i = 0; i < baneados.length; i++) { if (String(baneados[i].mza).trim() === mzaN && String(baneados[i].casa).trim() === casaN) { return true; } } return false; } /** * Devuelve todos los baneados activos de la hoja BANEADOS. * @returns {Array} [{mza, casa, id, motivo, ts}] */ function obtenerBaneados() { var hoja = obtenerHojaBaneados(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < 2) return []; var rows = hoja.getRange(2, 1, ultimaFila - 1, 6).getValues(); var baneados = []; for (var i = 0; i < rows.length; i++) { var mza = (rows[i][0] || '').toString().trim(); var casa = (rows[i][1] || '').toString().trim(); if (!mza || !casa) continue; // fila vacía var tsVal = rows[i][5]; var ts = (tsVal instanceof Date) ? tsVal.getTime() : (parseInt(tsVal) || 0); baneados.push({ mza: mza, casa: casa, id: (rows[i][2] || '').toString().trim(), motivo: (rows[i][3] || '').toString().trim(), ts: ts }); } return baneados; } /** * Graba un nuevo baneo en la hoja BANEADOS. * Si la Mza+Casa ya existe, actualiza el registro. */ function grabarBaneo(data) { var hoja = obtenerHojaBaneados(); var mzaN = String(data.mza || '').trim(); var casaN = String(data.casa || '').trim(); if (!mzaN || !casaN) return; // Buscar si ya existe var ultimaFila = hoja.getLastRow(); var filaExistente = -1; if (ultimaFila >= 2) { var rows = hoja.getRange(2, 1, ultimaFila - 1, 2).getValues(); for (var i = 0; i < rows.length; i++) { if (String(rows[i][0]).trim() === mzaN && String(rows[i][1]).trim() === casaN) { filaExistente = i + 2; break; } } } var ahora = new Date(); var fechaStr = Utilities.formatDate(ahora, 'GMT-3', 'dd/MM/yyyy HH:mm'); var ts = ahora.getTime(); var fila = (filaExistente !== -1) ? filaExistente : hoja.getLastRow() + 1; hoja.getRange(fila, 1).setValue(mzaN); hoja.getRange(fila, 2).setValue(casaN); hoja.getRange(fila, 3).setValue(data.id || ''); hoja.getRange(fila, 4).setValue(data.motivo || ''); hoja.getRange(fila, 5).setValue(fechaStr); hoja.getRange(fila, 6).setValue(ts); // Estilo visual: fondo rojo oscuro hoja.getRange(fila, 1, 1, 6) .setBackground('#c0392b') .setFontColor('white') .setFontWeight('bold'); SpreadsheetApp.flush(); } /** * Elimina el baneo de una Mza+Casa. */ function eliminarBaneo(mza, casa) { var hoja = obtenerHojaBaneados(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < 2) return; var mzaN = String(mza || '').trim(); var casaN = String(casa || '').trim(); var rows = hoja.getRange(2, 1, ultimaFila - 1, 2).getValues(); for (var i = rows.length - 1; i >= 0; i--) { if (String(rows[i][0]).trim() === mzaN && String(rows[i][1]).trim() === casaN) { hoja.deleteRow(i + 2); } } SpreadsheetApp.flush(); } /** * Devuelve la hoja BANEADOS, creándola si no existe. */ function obtenerHojaBaneados() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(NOMBRE_BANEADOS); if (!hoja) { hoja = ss.insertSheet(NOMBRE_BANEADOS); var encabezados = [['MZA','CASA','ID_RESIDENTE','MOTIVO','FECHA_BAN','TIMESTAMP']]; hoja.getRange(1, 1, 1, 6).setValues(encabezados); hoja.getRange(1, 1, 1, 6) .setFontWeight('bold') .setBackground('#6c2b2b') .setFontColor('white'); hoja.setColumnWidth(1, 80); hoja.setColumnWidth(2, 80); hoja.setColumnWidth(3, 140); hoja.setColumnWidth(4, 200); hoja.setColumnWidth(5, 140); hoja.hideColumns(6); // Oculta el timestamp (auxiliar) } return hoja; } // ── BÚSQUEDA HISTÓRICA ───────────────────────────────────── function buscarHistorico(fecha, q) { var hoja = obtenerHoja(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < FILA_INICIO) return []; var rows = hoja.getRange(FILA_INICIO, 1, ultimaFila - FILA_INICIO + 1, COL_ID).getValues(); var resultados = []; var fechaBuscar = fecha.trim(); var qBuscar = q.trim().toUpperCase(); for (var i = 0; i < rows.length; i++) { var patente = (rows[i][COL_PATENTE - 1] || '').toString().trim(); if (!patente) continue; var fechaVal = rows[i][COL_FECHA - 1]; var fechaStr = ''; var fechaISO = ''; try { if (fechaVal instanceof Date) { fechaStr = Utilities.formatDate(fechaVal, 'GMT-3', 'dd/MM/yyyy'); fechaISO = Utilities.formatDate(fechaVal, 'GMT-3', 'yyyy-MM-dd'); } } catch(e) {} if (fechaBuscar && fechaISO !== fechaBuscar) continue; var dni = (rows[i][COL_DNI - 1] || '').toString().trim(); if (qBuscar) { var matchPat = patente.toUpperCase().includes(qBuscar); var matchDni = dni.includes(qBuscar); if (!matchPat && !matchDni) continue; } var hEntrada = (rows[i][COL_ENTRADA - 1] || '').toString(); if (hEntrada.indexOf('\n') !== -1) hEntrada = hEntrada.split('\n')[1] || ''; var hSalida = (rows[i][COL_SALIDA - 1] || '').toString(); if (hSalida.indexOf('\n') !== -1) hSalida = hSalida.split('\n')[1] || ''; if (hSalida.indexOf('OK') === -1 && hSalida.indexOf(':') === -1) hSalida = ''; resultados.push({ pat: patente, dni: dni, mza: (rows[i][COL_MZA - 1] || '').toString(), casa: (rows[i][COL_CASA - 1] || '').toString(), tipo: (rows[i][COL_TIPO - 1] || '').toString().trim().toUpperCase(), hE: hEntrada, hS: hSalida, fecha: fechaStr }); } resultados.sort(function(a, b) { return (b.fecha + b.hE) > (a.fecha + a.hE) ? 1 : -1; }); return resultados; } // ── PÁNICO ───────────────────────────────────────────────── function grabarPanico(data) { var hoja = obtenerHojaPanico(); var fila = hoja.getLastRow() + 1; var ahora = new Date(); var id = ahora.getTime(); hoja.getRange(fila, 1).setValue(id); hoja.getRange(fila, 2).setValue(ahora).setNumberFormat("dd/MM/yyyy HH:mm:ss"); hoja.getRange(fila, 3).setValue(data.mza || ""); hoja.getRange(fila, 4).setValue(data.casa || ""); hoja.getRange(fila, 5).setValue("PENDIENTE"); hoja.getRange(fila, 6).setValue(data.mapsUrl || ""); hoja.getRange(fila, 7).setValue(data.lat || ""); hoja.getRange(fila, 8).setValue(data.lng || ""); hoja.getRange(fila, 1, 1, 8) .setBackground("#e74c3c") .setFontColor("white") .setFontWeight("bold"); SpreadsheetApp.flush(); } function obtenerPanicoPendiente() { var hoja = obtenerHojaPanico(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < 2) return []; var rows = hoja.getRange(2, 1, ultimaFila - 1, 8).getValues(); var pendientes = []; for (var i = 0; i < rows.length; i++) { var estado = (rows[i][4] || "").toString().trim(); if (estado !== "PENDIENTE") continue; var fechaVal = rows[i][1]; var hora = ""; try { hora = fechaVal instanceof Date ? Utilities.formatDate(fechaVal, "GMT-3", "HH:mm") : ""; } catch(e) {} pendientes.push({ panicoId: rows[i][0], fecha: hora, mza: (rows[i][2] || "").toString(), casa: (rows[i][3] || "").toString(), mapsUrl: (rows[i][5] || "").toString(), lat: (rows[i][6] || "").toString(), lng: (rows[i][7] || "").toString() }); } return pendientes; } function confirmarPanico(panicoId) { var hoja = obtenerHojaPanico(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < 2) return; var ids = hoja.getRange(2, 1, ultimaFila - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(panicoId)) { var fila = i + 2; hoja.getRange(fila, 5).setValue("VISTO"); hoja.getRange(fila, 1, 1, 8) .setBackground("#95a5a6") .setFontColor("white"); break; } } SpreadsheetApp.flush(); } function obtenerHojaPanico() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(NOMBRE_PANICO); if (!hoja) { hoja = ss.insertSheet(NOMBRE_PANICO); hoja.getRange(1, 1, 1, 8).setValues([["ID","FECHA","MZA","CASA","ESTADO","MAPS_URL","LAT","LNG"]]); hoja.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#2c3e50").setFontColor("white"); hoja.hideColumns(1); } return hoja; } // ── REGISTROS NORMALES ───────────────────────────────────── function obtenerActivos() { var hoja = obtenerHoja(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < FILA_INICIO) return []; var rows = hoja.getRange(FILA_INICIO, 1, ultimaFila - FILA_INICIO + 1, COL_ID).getValues(); var activos = []; for (var i = 0; i < rows.length; i++) { var patente = (rows[i][COL_PATENTE - 1] || "").toString().trim(); var salida = (rows[i][COL_SALIDA - 1] || "").toString().trim(); if (!patente || salida !== "") continue; var fechaVal = rows[i][COL_FECHA - 1]; var tsE = (fechaVal instanceof Date) ? fechaVal.getTime() : 0; var hEntrada = (rows[i][COL_ENTRADA - 1] || "").toString(); if (hEntrada.indexOf("\n") !== -1) hEntrada = hEntrada.split("\n")[1] || ""; activos.push({ id: rows[i][COL_ID - 1] || (FILA_INICIO + i), pat: patente, mza: (rows[i][COL_MZA - 1] || "").toString(), casa: (rows[i][COL_CASA - 1] || "").toString(), tipo: (rows[i][COL_TIPO - 1] || "VISITA").toString().trim().toUpperCase(), hE: hEntrada, hS: "", tsE: tsE }); } return activos; } function escribirEntrada(data) { var hoja = obtenerHoja(); var filaDestino = buscarFilaLibre(hoja); var ahora = new Date(); hoja.getRange(filaDestino, 1, 1, COL_ID) .setFontColor("black").setBackground(null).setFontWeight("normal"); hoja.getRange(filaDestino, COL_FECHA) .setValue(ahora).setNumberFormat("dd/MM/yyyy HH:mm"); hoja.getRange(filaDestino, COL_TIPO) .setValue((data.tipo || "VISITA").toUpperCase()); hoja.getRange(filaDestino, COL_MZA) .setValue(data.mza || ""); hoja.getRange(filaDestino, COL_CASA) .setValue(data.casa || ""); hoja.getRange(filaDestino, COL_DNI) .setValue(data.dni || ""); hoja.getRange(filaDestino, COL_PATENTE).setValue("'" + (data.pat || "").toUpperCase()); var horaStr = Utilities.formatDate(ahora, "GMT-3", "HH:mm"); hoja.getRange(filaDestino, COL_ENTRADA) .setValue("OK ✅\n" + horaStr) .setBackground("#d9ead3").setFontWeight("bold") .setHorizontalAlignment("center").setVerticalAlignment("middle").setWrap(true); if (data.id) hoja.getRange(filaDestino, COL_ID).setValue(data.id); SpreadsheetApp.flush(); } function marcarSalida(data) { var hoja = obtenerHoja(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < FILA_INICIO) return; var rows = hoja.getRange(FILA_INICIO, 1, ultimaFila - FILA_INICIO + 1, COL_ID).getValues(); var filaEncontrada = -1; if (data.id) { for (var i = rows.length - 1; i >= 0; i--) { if (String(rows[i][COL_ID - 1]) === String(data.id)) { filaEncontrada = FILA_INICIO + i; break; } } } if (filaEncontrada === -1) { var patenteBuscada = (data.pat || "").toUpperCase().trim(); for (var j = rows.length - 1; j >= 0; j--) { var patenteHoja = (rows[j][COL_PATENTE - 1] || "").toString().toUpperCase().trim(); var salidaHoja = (rows[j][COL_SALIDA - 1] || "").toString().trim(); if (patenteHoja === patenteBuscada && salidaHoja === "") { filaEncontrada = FILA_INICIO + j; break; } } } if (filaEncontrada === -1) return; var ahora = new Date(); var horaStr = Utilities.formatDate(ahora, "GMT-3", "HH:mm"); hoja.getRange(filaEncontrada, COL_SALIDA) .setValue("OK ✅\n" + horaStr) .setBackground("#ea9999").setFontWeight("bold") .setHorizontalAlignment("center").setVerticalAlignment("middle").setWrap(true); hoja.getRange(filaEncontrada, COL_FECHA, 1, COL_ID) .setBackground(null).setFontColor("black"); SpreadsheetApp.flush(); } // ── HELPERS ──────────────────────────────────────────────── function obtenerHoja() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(NOMBRE_HOJA); if (!hoja) hoja = ss.getSheets()[0]; return hoja; } function buscarFilaLibre(hoja) { var ultimaFila = hoja.getLastRow(); if (ultimaFila < FILA_INICIO) return FILA_INICIO; var valores = hoja.getRange(FILA_INICIO, COL_PATENTE, ultimaFila - FILA_INICIO + 2, 1).getValues(); for (var i = 0; i < valores.length; i++) { if (!valores[i][0]) return FILA_INICIO + i; } return ultimaFila + 1; } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ── MENÚ ─────────────────────────────────────────────────── function onOpen() { SpreadsheetApp.getUi() .createMenu("⚙️ NEXOS") .addItem("Abrir Panel de Control", "abrirPanelNexos") .addItem("Resumen del día", "resumenTurnoPopUp") .addItem("Ver baneados", "verBaneadosPopUp") .addToUi(); } function abrirPanelNexos() { try { var html = HtmlService.createHtmlOutputFromFile("Botonera") .setWidth(1200).setHeight(70) .setSandboxMode(HtmlService.SandboxMode.IFRAME); SpreadsheetApp.getUi().showModelessDialog(html, " "); } catch (e) { SpreadsheetApp.getUi().alert("Panel externo activo.\nUsá la URL del Web App desde el celular."); } } function resumenTurnoPopUp() { var hoja = obtenerHoja(); var ui = SpreadsheetApp.getUi(); var ultimaFila = hoja.getLastRow(); if (ultimaFila < FILA_INICIO) { ui.alert("No hay datos."); return; } var datos = hoja.getRange(FILA_INICIO, 1, ultimaFila - FILA_INICIO + 1, COL_SALIDA).getValues(); var ahora = new Date(); var hoyStr = Utilities.formatDate(ahora, "GMT-3", "dd/MM/yyyy"); var entradas = 0, salidas = 0, adentro = {}; datos.forEach(function(fila) { var fechaVal = fila[COL_FECHA - 1]; var patente = (fila[COL_PATENTE - 1] || "").toString().trim().toUpperCase(); var salidaV = (fila[COL_SALIDA - 1] || "").toString().trim(); if (!patente) return; var fechaStr = (fechaVal instanceof Date) ? Utilities.formatDate(fechaVal, "GMT-3", "dd/MM/yyyy") : ""; if (fechaStr === hoyStr) { entradas++; if (salidaV !== "") { salidas++; } else { adentro[patente] = true; } } }); var lista = Object.keys(adentro); var msg = "📥 ENTRADAS: " + entradas + "\n📤 SALIDAS: " + salidas + "\n🚗 ADENTRO: " + lista.length + "\n──────────────────\n" + (lista.length ? "• " + lista.join("\n• ") : "Ninguno"); ui.alert("📊 RESUMEN DEL DÍA (" + hoyStr + ")", msg, ui.ButtonSet.OK); } /** * Muestra un popup con los vecinos baneados actualmente. */ function verBaneadosPopUp() { var ui = SpreadsheetApp.getUi(); var baneados = obtenerBaneados(); if (!baneados.length) { ui.alert("🚷 BANEADOS", "No hay vecinos baneados actualmente.", ui.ButtonSet.OK); return; } var lineas = baneados.map(function(b) { return "Mza " + b.mza + " — Casa " + b.casa + (b.id ? " (ID: " + b.id + ")" : "") + (b.motivo ? "\n ⚠️ " + b.motivo : ""); }); ui.alert( "🚷 VECINOS BANEADOS (" + baneados.length + ")", lineas.join("\n─────────────────\n"), ui.ButtonSet.OK ); }
